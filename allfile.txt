Understanding K:
    Compile into: 
        x86 or arm64 asm
    File extensions:
        .k, .kl, .kl
    Purpose:
        Low level programming
    Compiler name:
        kcc
    Flags:
        -bo (exe or appimage depends on system),
        -ffocus (full compile speed),
        -default (disable std library),
        -derr (acts like -Wall on gcc),
        -ff (goes into freestanding mod),
        -r (builds and runs),
    Comments: 
        ;;  <- thats a Comment
    Multi Comment:
        ;*     *;
    Memory managment:
        if Safe is imported use a gc (garbage collecter) 
        if not use manual memory:
            malloc, calloc, free etc

Syntax:
    imports noted as:
        $unpack-> <File>;
    
    main (stdlib):
        $unpack-> <Safe>;
        ;; This imports the full std lib;

        $entry-> <main>;
        ;; Sets a main entry
        ;; Can be changed to anything

        init::exec main[safe::nul] -> i32 {
            ;; initializes a function named main with no arguments, and returns a i32


            ret 0;
            ;; return 0
        }

    printing + variables + data types:
        $unpack-> <Safe>;

        $entry-> <main>;

        init::exec main[safe::nul] -> i32 {
            safe::print["Hello World\n"];
            ;; prints hello World

            def exp1: i64 = 5075;
            ;* Can be:
                i8, i16, i32
            *;

            def exp2: f64 = 7600.55;
            ;* Can be:
                f8, f16, f32
            *;

            def exp3: safe::str = "Alex";
            ;* Just a string...
            *;

            def exp4: byte = 'A';
            ;* Can be:
                char
            *;

            def exp5: bool = true;

            def exp6: u64 = 7500000;
            ;* Can be:
                u8, u16, u32
            *;

            ;; printing everything:
                safe::print["%d\n", exp1];
                safe::print["%dd\n", exp2];
                safe::print["%s\n", exp3];
                safe::print["%c\n", exp4];
                safe::print["%b\n", exp5];
                safe::print["%u\n", exp6];

            ;; variables are inmutable by default;
            ;; to mutate them:
                ;; add this:
            def x: mut, i8 = 200;

            ;; global variables:
                def y: pub, mut, i16 = 670;
            ;; all variable are static!!!!



            ret 0;
        }

    arithemetic:
        signs:
            +, -, *, /
            ** (power)
            % (module)
            ++ (+ 1)
            -- (- 1)

        operational signs:
            ||, &&, !
            !=, ==, <, >, =<, >=
    flow control:
        if else:
            ifd (x < y)>>do {
                safe::print["bigger\n"];
            }
            else ifd (x == y)>>do {
                safe::print["equal\n"];
            }
            else>>do {
                safe::print["smaller\n"];
            }
        while, loops:
            for (def i: i64 = 0; i <= 1000000; i++)>>do {
                safe::print["%d\n", i];
            }

            while(1 == 1)>>do {
                safe::print["forever loop\n"];
            }
    Input + namespaces:
        to get input:
            safe::getin["%d", x];
        
        to put a name on a data type etc do:
            set i32, int;
    Structures:
        struct Example {
            x: safe::str;
            y: i64;
            z: char;
        }
        calling structures:'
            init::exec examplexd[Example *exp] -> safe::nul {
                &exp->x = "Hello";
                &exp->y = 8000;
                &exp->z = 'A';
            }

        enums:
            list::enum Example {
                apple,
                banana,
                grape,
                strawberry
            };

            to call an enum:
                Example@apple, Example@banana, Example@grape, Example@strawberry;
                    Must use '@' to call objects in a enum!
                            or you can make variables have enum types:
                                def xzy: mut, Example, pub;

                                xzy = apple;
                                xzy = grape;
                                    etc...

            arrays:
                def array_example: mut, i64, ar[25];
                def array_example1: mut, safe::str ar[5] = { "Windows", "Cachy OS" etc..   };

            fixed float points:
                f8, f16, f32, f64...
                    or
                    double = f64;
                


        kernel, freestanding:
            built in asm:
                core::asm["hlt"];

            add exact width:
                u8, u16, u32, u64;
                i8, i16, i32, i64;
            add type for low level strings:
                byte* or char*;
        Make it able to touch, read, write on memory addresses.
        pointers for variables:
            def i: ptr, mut, pub, u32 = 8000;
            def ptr* = i;
